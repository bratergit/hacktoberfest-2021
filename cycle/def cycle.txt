def cycle_sort(arr):
    n = len(arr)

    for cycle_start in range(n - 1):
        item = arr[cycle_start]

        # Find the correct position of item
        pos = cycle_start
        for i in range(cycle_start + 1, n):
            if arr[i] < item:
                pos += 1

        # Already in correct position
        if pos == cycle_start:
            continue

        # Skip duplicates
        while item == arr[pos]:
            pos += 1

        # Put item in correct position
        arr[pos], item = item, arr[pos]

        # Rotate the cycle
        while pos != cycle_start:
            pos = cycle_start

            # Find correct position
            for i in range(cycle_start + 1, n):
                if arr[i] < item:
                    pos += 1

            # Skip duplicates
            while item == arr[pos]:
                pos += 1

            # Swap
            arr[pos], item = item, arr[pos]

    return arr


# Example
arr = [20, 40, 50, 10, 30]

print("Before sorting:", arr)

cycle_sort(arr)

print("After sorting:", arr)
